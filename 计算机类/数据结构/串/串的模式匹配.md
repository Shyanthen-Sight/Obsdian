### BF算法
```c

//注意：代码是**1 号位起始**的顺序串 `SString`（字符数组 `ch[1]` 存第一个字符，`ch[0]` 闲置）。
int Index(SString S, SString T){
	int i = 1, j = 1;
	while(i <= S.length && j <= T.length){
		if(S.ch[i] == T.ch[j]){
			++i; ++j;	//继续比较后继字符
		}else{
			//指针后退重新开始匹配
			i = i-j+2;
			j = 1;
		}
	}
	if(j > T.length){
		return i - T.length;
	}
	else
	{
		return 0;
	}
}
```
