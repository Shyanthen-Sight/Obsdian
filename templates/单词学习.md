// 生成当天基础名，如 E26-10-4；若已存在则依次追加 -2、-3……
let base = "E" + tp.date.now("YY-M-D");
let folder = tp.file.folder(true);                       // 完整路径，如 "英语/单词学习"；根目录为 "/"
let prefix = (folder && folder !== "/") ? folder + "/" : "";
let name = base, n = 2;
while (await tp.file.exists(prefix + name + ".md")) { name = base + "-" + n; n++; }
await tp.file.rename(name);
%>
