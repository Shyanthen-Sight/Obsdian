
### 一.数值函数

```sql
SELECT ABS(-123)                 #取绝对值

SELECT CEIL(2.3),CEIL(-2.3);     #向上取整

SELECT FLOOR(5.2),FLOOR(-5.2);   #向下取整

SELECT RAND();                   #随机数

SELECT ROUND(2.3,0);             #四舍五入

SELECT ROUND(2.36,1);            #四舍五入并保留一位

SELECT TRUNCATE(8.97,1);         #直接保留一位小数
```

### 二.字符串函数

```sql
SELECT CHAR_LENGTH('hello');               #返回字符串长度

SELECT LENGTH('hello');                    #返回字符串的字节数

SELECT CONCAT('how ','are ','you?');       #字符串的连接

SELECT FIND_IN_SET('usgo','abc,usgo,test');#返回字符串在字符串组中的出现的位置

SELECT REPLACE(原字符串, 要替换的值, 替换成的值)
```

### 三.时间函数

```sql
SELECT CURDATE();#显示日期

SELECT NOW();#显示日期和时间

SELECT YEAR('2006-11-26');#显示年份，month（），day（）一样

SELECT MONTHNAME('2006-11-26');#显示月份名称

SELECT QUARTER('2006-11-26');#返回季度，范围为1-4

SELECT DAYOFYEAR('2006-11-26');#返回一年中的第几天

SELECT DAYOFMONTH('2006-11-26');#返回一个月中的第几天

SELECT WEEK('2006-11-26');#返回一年中的第几周

SELECT WEEKDAY('2026-6-1');#返回周几

SELECT DATE_ADD('2006-11-26',INTERVAL 2 MONTH);#原时间基础上加2个月

SELECT DATE_ADD('2006-11-26',INTERVAL 2 DAY);

SELECT DATE_SUB('2006-11-26',INTERVAL 2 MONTH);#原时间基础上减2个月
SELECT DATE_SUB('2006-11-26',INTERVAL 2 DAY);

SELECT DATEDIFF('2006-11-26','2006-11-23');#(noly)计算天数相差多少
SELECT SUBTIME('11:00','11:30');#前者-后者=时间差
```
